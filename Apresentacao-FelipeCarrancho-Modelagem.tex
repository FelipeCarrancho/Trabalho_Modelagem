\section{Problema e Modelagem}
%~~~~~~~~~~~~~~~~~~~~~~~~~~

%------------------------------------------------
\begin{frame}{Sistema estudado}\includelogo
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Modelo físico}
Sistema massa-mola-amortecedor com inércia, dissipação viscosa e armazenamento elástico.
\end{block}
\vspace{0.3em}
\begin{itemize}
  \item Massa: $m=8$ kg
  \item Amortecimento: $C=2$ N$\cdot$s/m
  \item Rigidez: $K=0{,}5$ N/m
  \item Entrada: força externa $f(t)$
\end{itemize}
\column{0.48\textwidth}
\centering
\includegraphics[width=0.70\linewidth,height=0.52\textheight,keepaspectratio]{imagem_sistema.jpeg}
\figrel{1}{Representação física do sistema massa-mola-amortecedor.}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Equação diferencial e espaço de estados}\includelogo
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Equação do movimento}
\smallformula{m\ddot{x}(t)+C\dot{x}(t)+Kx(t)=f(t)}
\smallformula{8\ddot{x}(t)+2\dot{x}(t)+0{,}5x(t)=f(t)}
\end{block}
\column{0.48\textwidth}
\begin{block}{Forma de primeira ordem}
\smallformula{\dot{x}(t)=v(t)}
\smallformula{\dot{v}(t)=\frac{f(t)-C v(t)-K x(t)}{m}}
\smallformula{\dot{v}(t)=0{,}125f(t)-0{,}25v(t)-0{,}0625x(t)}
\end{block}
\end{columns}
\vspace{0.2em}
\begin{alertblock}{Parâmetros dinâmicos}
\centering
$\omega_n=\sqrt{K/m}=0{,}25$ rad/s \hspace{1cm} $\zeta=\dfrac{C}{2\sqrt{mK}}=0{,}5$ \hspace{1cm} Sistema subamortecido
\end{alertblock}
\end{frame}

\section{Malha Fechada e Estabilidade}
%~~~~~~~~~~~~~~~~~~~~~~~~~~

%------------------------------------------------
\begin{frame}{Atividade 4 -- Diagrama de controle}\includelogo
\begin{columns}[T]

\figrel{9}{Diagrama de blocos do sistema de controle realimentado.}
\column{0.38\textwidth}
\begin{block}{Estrutura analisada}
\begin{itemize}
  \item Controlador proporcional $G_c(s)=K$.
  \item Planta $G_p(s)=1/(8s^2+2s+0{,}5)$.
  \item Sensor $H(s)=\frac{4}{3}s+1$.
  \item Realimentação negativa.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Atividade 4 -- Controle proporcional e sensor}\includelogo
\begin{columns}[T]
\column{0.52\textwidth}
\begin{block}{Malha fechada}
\smallformula{\frac{C(s)}{R(s)}=\frac{G_c(s)G_p(s)}{1+G_c(s)G_p(s)H(s)}}
\smallformula{G_c(s)=K=\frac{8}{3},\quad H(s)=\frac{4}{3}s+1}
\smallformula{\frac{C(s)}{R(s)}=\frac{64s+48}{192s^3+192s^2+48s+57}}
\end{block}
\column{0.44\textwidth}
\begin{block}{Routh-Hurwitz}
\centering
\begin{tabular}{c|cc}
$s^3$ & 192 & 48 \\
$s^2$ & 192 & 57 \\
$s^1$ & -9 & 0 \\
$s^0$ & 57 & 0
\end{tabular}
\end{block}
\begin{alertblock}{Resultado}
Mudança de sinal na primeira coluna: malha fechada instável para $K=8/3$.
\end{alertblock}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Atividade 4 -- Faixa de estabilidade}\includelogo
\begin{columns}[T]
\column{0.47\textwidth}
\begin{block}{Denominador genérico}
{\small
\[
D(s)=(8s^2+2s+0{,}5)\left(\frac{4}{3}s+1\right)+K
\]
}
\end{block}

\begin{alertblock}{Condição de estabilidade}
\centering
$\displaystyle -\frac{1}{2}<K<\frac{13}{6}$
\end{alertblock}

\begin{itemize}\small
  \item $K_c=8/3\approx2{,}667$.
  \item Limite superior: $13/6\approx2{,}167$.
  \item Como $8/3>13/6$, o ganho nominal fica fora da região estável.
\end{itemize}
